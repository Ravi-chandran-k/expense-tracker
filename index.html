function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const entriesSheet = ss.getSheetByName('Entries') || ss.insertSheet('Entries'); const budgetsSheet = ss.getSheetByName('Budgets') || ss.insertSheet('Budgets'); const rows = entriesSheet.getDataRange().getValues(); const entries = rows.slice(1).filter(r => r[0]).map(r => ({ id: r[0], date: r[1], category: r[2], amount: r[3], note: r[4] })); const raw = budgetsSheet.getRange('A1').getValue(); return ContentService.createTextOutput(JSON.stringify({ entries, budgets: raw ? JSON.parse(raw) : {} })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const entriesSheet = ss.getSheetByName('Entries') || ss.insertSheet('Entries'); const budgetsSheet = ss.getSheetByName('Budgets') || ss.insertSheet('Budgets'); if (entriesSheet.getLastRow() === 0) entriesSheet.appendRow(['ID','Date','Category','Amount','Note']); const data = JSON.parse(e.postData.contents); (data.entries || []).forEach(entry => { const rows = entriesSheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < rows.length; i++) if (rows[i][0] === entry.id) { rowIndex = i + 1; break; } const rowData = [entry.id, entry.date, entry.category, entry.amount, entry.note || '']; if (rowIndex > -1) entriesSheet.getRange(rowIndex, 1, 1, 5).setValues([rowData]); else entriesSheet.appendRow(rowData); }); (data.deletedIds || []).forEach(id => { const rows = entriesSheet.getDataRange().getValues(); for (let i = rows.length - 1; i >= 1; i--) if (rows[i][0] === id) { entriesSheet.deleteRow(i + 1); break; } }); if (data.budgets) budgetsSheet.getRange('A1').setValue(JSON.stringify(data.budgets)); return ContentService.createTextOutput(JSON.stringify({ status: 'ok' })).setMimeType(ContentService.MimeType.JSON); }
